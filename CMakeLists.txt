cmake_minimum_required(VERSION 3.20)
project(tmux_native_port VERSION 0.0.1 LANGUAGES C)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This tree builds tmux for Windows with Visual Studio "
    "(MSVC) only. On other systems, build tmux itself: "
    "https://github.com/tmux/tmux")
endif()
include(cmake/Version.cmake)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
if(MSVC)
  add_compile_options(/utf-8)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(TMUX_WIN32_MIN_BUILD "17763" CACHE STRING "Minimum Windows build (17763 = Windows 10 1809, the ConPTY floor)")
if(TMUX_WIN32_MIN_BUILD LESS 17763)
  message(FATAL_ERROR "TMUX_WIN32_MIN_BUILD below 17763: ConPTY does not exist before Windows 10 1809")
endif()
foreach(pair "26100;0x0A000010" "22621;0x0A00000C" "22000;0x0A00000B" "20348;0x0A00000A"
             "19041;0x0A000008" "18362;0x0A000007" "17763;0x0A000006")
  list(GET pair 0 build)
  list(GET pair 1 ntddi)
  if(NOT TMUX_WIN32_MIN_BUILD LESS build)
    break()
  endif()
endforeach()
add_compile_definitions(_WIN32_WINNT=0x0A00 WINVER=0x0A00 NTDDI_VERSION=${ntddi}
  TMUX_WIN32_MIN_BUILD=${TMUX_WIN32_MIN_BUILD} WIN32_LEAN_AND_MEAN NOMINMAX)
include(cmake/Windows.cmake)
add_subdirectory(platform)
target_link_libraries(tmux PRIVATE tmux-platform)
if(TMUX_WIN32_CONPTY_BUNDLE)
  target_compile_definitions(tmux-platform PRIVATE TMUX_WIN32_CONPTY_DEFAULT="auto")
endif()
